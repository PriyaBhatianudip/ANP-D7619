Password Strength Checker
-> password should contain at least 1 upper, 1smaller, 1 special character, and 1 digit
string = "Hello@123*"
upper= 0
lower=0
digits=0
specials =0

for ch in string:
	if "A" <= ch <="Z":
		upper+=1
	elif "a" <= ch <="z":
		lower+=1
	elif "0" <= ch <="9":
		digits+=1
	else:
		specials+=1

if upper>0 and lower>0 and digits>0 and specials>0 and
	len(string)>=6:
	print("Password strength is good")
	return True
else:
	print("Invalid password")
	return False

-> the length of the password should be at least 6 characters.
